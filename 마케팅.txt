function hintText(i){

  const c=currentMarket().contents;

  if(i===0){

    const best=[...c].sort((a,b)=>b[2]-a[2])[0];

    return `시장 관심도가 높은 콘텐츠는 "${best[0]}"입니다.`;

  }

  if(i===1){

    const best=[...c].sort((a,b)=>b[3]-a[3])[0];

    return `소비자 반응이 가장 강할 콘텐츠는 "${best[0]}"입니다.`;

  }

  if(i===2){

    return "경쟁사들이 글로벌 확장형 콘텐츠에 관심을 보이고 있습니다.";

  }

  return "새로운 조합과 글로벌 확장성을 가진 콘텐츠가 유리합니다.";

}


function buyHint(i){

  const h=hints[i];

  if(state.boughtHints.includes(i)) return;

  if(state.money<h.cost){

    alert("예산이 부족합니다.");

    return;

  }

  state.money-=h.cost;

  state.boughtHints.push(i);

  render();

}

function submitBid(){

  const aiBid =
    Math.floor(250 + Math.random()*500);

  if(state.bid>state.money){

    alert("예산보다 높은 금액은 제안할 수 없습니다.");

    return;

  }

  if(state.bid>=aiBid){

    state.acquired=true;

    state.money-=state.bid;

    alert(
      "콘텐츠 확보 성공!\n"+
      "경쟁사의 제안가: "+
      aiBid+"P"
    );

  }

  else{

    state.acquired=false;

    alert(
      "콘텐츠 확보 실패!\n"+
      "경쟁사의 제안가: "+
      aiBid+"P"
    );

  }

  go("marketing");

}

function getHitContent(){

  const contents=currentMarket().contents;

  let best=0;
  let bestScore=-Infinity;

  contents.forEach((c,i)=>{

    const score =
      c[2] +
      c[3] +
      Math.random()*20;

    if(score>bestScore){

      bestScore=score;
      best=i;

    }

  });

  return best;

}


function marketing(type){

  if(type==="sns"){

    state.money+=300;
    state.share+=8;
    state.image+=2;

  }

  if(type==="fandom"){

    state.money+=380;
    state.share+=12;
    state.image+=5;

  }

  if(type==="global"){

    state.money+=450;
    state.share+=15;
    state.image+=3;

  }

  if(type==="brand"){

    state.money+=350;
    state.share+=7;
    state.image+=12;

  }

  const hit=getHitContent();

  if(
    state.acquired &&
    state.selectedContent===hit
  ){

    state.money+=700;
    state.share+=15;

  }

  else if(state.acquired){

    state.money-=150;

  }

  state.event =
    events[Math.floor(Math.random()*events.length)];

  go("event");

}
